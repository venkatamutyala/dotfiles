# #!/bin/zsh # # dotfiles installer | https://github.com/venkatamutyala/dotfiles # Served at https://setup.venkatamutyala.com # # This file is a polyglot -- it works two ways: # * Install: curl setup.venkatamutyala.com | zsh # * Read it: open https://setup.venkatamutyala.com in a browser. The # "#<plaintext>" marker on line 1 tells the browser to render # the rest of this file as raw monospace text (no HTML parsing, # whitespace preserved). zsh just treats that line as a comment, # so piping it to zsh still runs the installer normally. # # Linux/Debian only, used mostly over SSH / remote VMs / devcontainers. # Installs Oh-My-Zsh + zsh plugins + config files. Does NOT install system # packages -- install tmux/gh/jq/kubectl/nethogs/vim yourself (see README). # Safe to re-run; backs up anything it would overwrite. set -euo pipefail # Where to fetch config files from. Override for testing a fork/branch/local # checkout, e.g. REPO_RAW="file:///repo" (used by the CI integration test). REPO_RAW="${REPO_RAW:-https://raw.githubusercontent.com/venkatamutyala/dotfiles/refs/heads/main}" TS="$(date +%Y%m%d-%H%M%S)" BACKUP_DIR="${HOME}/.dotfiles-backup-${TS}" log() { print -r -- "==> $*"; } warn() { print -r -- "WARN: $*" >&2; } die() { print -r -- "ERROR: $*" >&2; exit 1; } os_guard() { [[ "$(uname -s)" == "Linux" ]] || die "Linux (Debian/Ubuntu) only. Detected $(uname -s). Aborting." if [[ -r /etc/os-release ]]; then local id idlike id="$(. /etc/os-release 2>/dev/null; print -r -- "${ID:-}")" idlike="$(. /etc/os-release 2>/dev/null; print -r -- "${ID_LIKE:-}")" [[ "$id" == debian || "$id" == ubuntu || "$idlike" == *debian* ]] \ || warn "Non Debian/Ubuntu Linux (ID='$id'); continuing but untested." fi command -v git >/dev/null 2>&1 || die "git is required. Install it and re-run." command -v curl >/dev/null 2>&1 || die "curl is required. Install it and re-run." } backup_one() { # copy (not move) existing target into timestamped backup local t="$1"; [[ -e "$t" || -L "$t" ]] || return 0 mkdir -p "$BACKUP_DIR" local dest="${BACKUP_DIR}/${t#$HOME/}" mkdir -p "${dest:h}"; cp -a "$t" "$dest"; log " backed up: $t" } backup_existing() { log "Backing up any existing files to ${BACKUP_DIR}" for f in ~/.tmux.conf ~/.vimrc ~/.config/htop/htoprc ~/.zshrc ~/.claude/settings.json \ ~/.oh-my-zsh/custom/vm-manage.sh ~/.oh-my-zsh/custom/venkatamutyala-functions.zsh; do backup_one "$f" done } install_omz() { # non-interactive; never clobbers .zshrc or login shell if [[ -d ~/.oh-my-zsh ]]; then log "Oh-My-Zsh already present; skipping." else log "Installing Oh-My-Zsh (non-interactive)" RUNZSH=no CHSH=no KEEP_ZSHRC=yes \ sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)" "" --unattended fi [[ -f ~/.zshrc ]] || cp ~/.oh-my-zsh/templates/zshrc.zsh-template ~/.zshrc } fetch() { local u="$1" d="$2"; mkdir -p "${d:h}"; curl -fsSL "$u" -o "$d"; log " fetched: $d"; } fetch_configs() { log "Fetching configs and custom scripts" mkdir -p ~/.oh-my-zsh/custom fetch "${REPO_RAW}/vm-manage.sh" ~/.oh-my-zsh/custom/vm-manage.sh chmod +x ~/.oh-my-zsh/custom/vm-manage.sh fetch "${REPO_RAW}/venkatamutyala-functions.zsh" ~/.oh-my-zsh/custom/venkatamutyala-functions.zsh fetch "${REPO_RAW}/tmux.conf" ~/.tmux.conf fetch "${REPO_RAW}/htoprc" ~/.config/htop/htoprc fetch "${REPO_RAW}/vimrc" ~/.vimrc fetch "${REPO_RAW}/claude-settings.json" ~/.claude/settings.json fetch "https://raw.githubusercontent.com/junegunn/vim-plug/master/plug.vim" ~/.vim/autoload/plug.vim } configure_zshrc_plugins() { # idempotent; handles single- & multi-line plugins=() local zshrc=~/.zshrc local desired='plugins=(git zsh-autosuggestions kubectl zsh-syntax-highlighting you-should-use)' [[ -f "$zshrc" ]] || { warn "~/.zshrc missing; skipping plugins."; return 0; } if grep -Eq '^[[:space:]]*plugins=\(' "$zshrc"; then awk -v repl="$desired" ' done==1 { print; next } /^[[:space:]]*plugins=\(/ { print repl; if ($0 ~ /\)/) {done=1;next} inblock=1; next } inblock==1 { if ($0 ~ /\)/) {inblock=0;done=1} next } { print }' "$zshrc" > "${zshrc}.tmp" && mv "${zshrc}.tmp" "$zshrc" elif grep -q 'oh-my-zsh.sh' "$zshrc"; then awk -v repl="$desired" '/source .*oh-my-zsh\.sh/ && !ins {print repl; ins=1} {print}' \ "$zshrc" > "${zshrc}.tmp" && mv "${zshrc}.tmp" "$zshrc" else print -r -- "$desired" >> "$zshrc" fi log "Configured plugins=() in ~/.zshrc" } clone_plugin() { # idempotent; tolerates already-installed under set -e local url="$1" ref="$2" dir="$3" [[ -d "$dir/.git" ]] && { log " plugin present: ${dir:t}"; return 0; } [[ -e "$dir" ]] && { backup_one "$dir"; rm -rf "$dir"; } git -c advice.detachedHead=false clone --depth 1 --branch "$ref" "$url" "$dir"; log " cloned: ${dir:t}@${ref}" } install_zsh_plugins() { log "Installing zsh plugins" local p=~/.oh-my-zsh/custom/plugins; mkdir -p "$p" clone_plugin https://github.com/zsh-users/zsh-syntax-highlighting 0.8.0 "$p/zsh-syntax-highlighting" clone_plugin https://github.com/zsh-users/zsh-autosuggestions v0.7.1 "$p/zsh-autosuggestions" clone_plugin https://github.com/MichaelAquilina/zsh-you-should-use.git 1.10.0 "$p/you-should-use" } install_vim_plugins() { if command -v vim >/dev/null 2>&1; then log "Installing vim plugins (vim +PlugInstall +qall)" vim -E -s -u ~/.vimrc +PlugInstall +qall </dev/null >/dev/null 2>&1 \ || warn "vim +PlugInstall returned non-zero (often harmless); re-run 'vim +PlugInstall +qall' if needed." else warn "vim not found; skipping. After installing vim, run: vim +PlugInstall +qall" fi } main() { os_guard backup_existing install_omz fetch_configs configure_zshrc_plugins install_zsh_plugins install_vim_plugins log "Done. Backups (if any): ${BACKUP_DIR}" print -r -- "" print -r -- "Next: start a fresh shell ('exec zsh'); reload tmux with 'tmux source-file ~/.tmux.conf'." print -r -- "Install deps yourself if missing: tmux(>=3.2) gh jq kubectl nethogs vim" } main "$@"
